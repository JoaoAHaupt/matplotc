43void draw_linear_correlation(Data *data, int data_count, Plane *plane,
44 PlotArea *plot) {
45 float x_range = plane->x_min_max[1] - plane->x_min_max[0];
46 float y_range = plane->y_min_max[1] - plane->y_min_max[0];
47
48 if (x_range <= 0 || y_range <= 0)
49 return;
50
51 for (int j = 0; j < data_count; j++) {
52 int n = data[j].size;
53
54 if (n < 2)
55 continue;
56
57 double sum_x = 0;
58 double sum_y = 0;
59 double sum_xy = 0;
60 double sum_x2 = 0;
61
62 for (int i = 0; i < n; i++) {
63 double x = data[j].x[i];
64 double y = data[j].y[i];
65
66 sum_x += x;
67 sum_y += y;
68 sum_xy += x * y;
69 sum_x2 += x * x;
70 }
71
72 double denominator = n * sum_x2 - sum_x * sum_x;
73
74 if (fabs(denominator) < 1e-9)
75 continue;
76
77 double a = (n * sum_xy - sum_x * sum_y) / denominator;
78 double b = (sum_y - a * sum_x) / n;
79
80 double x1 = plane->x_min_max[0];
81 double x2 = plane->x_min_max[1];
82
83 double y1 = a * x1 + b;
84 double y2 = a * x2 + b;
85
86 float screen_x1 =
87 plot->left + (x1 - plane->x_min_max[0]) / x_range * plot->width;
88
89 float screen_y1 =
90 plot->bottom - (y1 - plane->y_min_max[0]) / y_range * plot->height;
91
92 float screen_x2 =
93 plot->left + (x2 - plane->x_min_max[0]) / x_range * plot->width;
94
95 float screen_y2 =
96 plot->bottom - (y2 - plane->y_min_max[0]) / y_range * plot->height;
97
98 DrawLineEx((Vector2){screen_x1, screen_y1}, (Vector2){screen_x2, screen_y2},
99 1.0f, COLOURS[j]);
100 }
