42void draw_axis_x(float scale, float min_max[2], bool lines,
43 PlotArea *plot_area) {
44 if (scale <= 0.0f) {
45 return;
46 }
47
48 int number_of_scale = (int)((min_max[1] - min_max[0]) / scale);
49
50 if (number_of_scale <= 0) {
51 return;
52 }
53
54 float scale_pixel_x = plot_area->width / number_of_scale;
55 float value = min_max[0];
56
57 for (int i = 0; i <= number_of_scale; i++) {
58 float x = plot_area->left + i * scale_pixel_x;
59
60 DrawRectangle((int)(x - 1), (int)(plot_area->bottom - (float)tick_size / 2),
61 2, tick_size, BLACK);
62
63 if (lines && i > 0 && i < number_of_scale) {
64 DrawLine((int)x, (int)plot_area->top, (int)x, (int)plot_area->bottom,
65 LIGHTGRAY);
66 }
67
68 const char *label = TextFormat("%g", value);
69 int text_width = MeasureText(label, font_size);
70
71 DrawText(label, (int)(x - (float)text_width / 2),
72 (int)(plot_area->bottom + 12), font_size, BLACK);
73
74 value += scale;
75 }
