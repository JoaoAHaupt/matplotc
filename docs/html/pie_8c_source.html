22static void draw_pie_legend(Data *data, int data_count, PlotArea *plotarea) {
23 float total = 0;
24 for (int j = 0; j < data_count; j++) {
25 for (int i = 0; i < data[j].size; i++) {
26 total += data[j].y[i];
27 }
28 }

40
41 float legend_x = plotarea->right - 150;
42 float legend_y = plotarea->top + 10;
43

45 for (int j = 0; j < data_count; j++) {

55 data[j].y[i] / total * 100.0f);
56 DrawTextEx(font, label,
57 (Vector2){legend_x + box_size + padding, y - 2}, font_size,
58 spacing, BLACK);

81 // calcula o angulo

83
84 DrawCircleSector((Vector2){400, 400}, 150, start_angle,
